\begin{frame}
\SlideHeader{02 \textperiodcentered\ OVERVIEW}{What PINK does, in one slide}

\begin{columns}[T]
\column{0.54\linewidth}
{\sffamily\footnotesize
PINK (Liu \textit{et al.}, \textit{J. Mater. Inf.} 2025) predicts \textbf{lattice
thermal conductivity} $\kappa_L$ from a CIF file, without solving the phonon
Boltzmann transport equation.

\vspace{4pt}
A graph network predicts two elastic moduli; a closed-form model turns them into
$\kappa_L$. That physics stage has \textbf{no parameters fitted to this data}
(slide 16).}

\vspace{5pt}
\Card{6.3cm}{CardBg}{CardRule}{%
  \sffamily\footnotesize
  \textbf{Why bother}\\[3pt]
  DFT + phonons take days per material; this takes about a sixth of a second
  --- enough to screen half a million.}

\column{0.42\linewidth}
{\sffamily\footnotesize\color{Muted} The two stages:}\\[4pt]
\begin{tikzpicture}[node distance=4mm,
  box/.style={rounded corners=4pt, fill=CardBg, draw=CardRule, line width=0.7pt,
              text width=4.5cm, align=left, inner sep=6pt,
              font=\sffamily\scriptsize\color{SlideInk}},
  arr/.style={font=\color{SlideBlue}\small}]
  \node[box] (b1) {\textbf{Stage 1 --- learned}\\ CIF $\rightarrow$ graph network
                   $\rightarrow$ bulk modulus $K$, shear modulus $G$};
  \node[arr, below=of b1] (a1) {$\downarrow$};
  \node[box, below=of a1] (b2) {\textbf{Stage 2 --- physics}\\ $K$, $G$
                   $\rightarrow$ Poisson ratio $\rightarrow$ sound velocity,
                   Debye temperature, Gr\"uneisen $\gamma$ $\rightarrow$ $\kappa_L$};
\end{tikzpicture}
\end{columns}

\Takeaway{The network never sees a thermal conductivity. It learns elasticity;
the physics does the rest.}
\end{frame}

\begin{frame}
\SlideHeader{06 \textperiodcentered\ ALIGNN}{Why a second architecture at all}

{\sffamily\footnotesize
A single model that scores well can be right for the wrong reason. Two models
that share the training data and the physics, but not the architecture, give a
cross-check: where they agree, the result is less likely to be an artefact of
either one.

\vspace{4pt}
ALIGNN was chosen because it differs from CGCNN in a \textbf{physically
motivated} way. It is not a bigger CGCNN: it sees something CGCNN cannot ---
\textbf{bond angles}.}

\vspace{6pt}
\begin{columns}[T]
\column{0.47\linewidth}
\Card{5.6cm}{CardBg}{CardRule}{%
  \sffamily\footnotesize
  \textbf{What CGCNN encodes}\\[3pt]
  Which atoms exist, which are bonded, and \emph{how far apart} they are.
  A two-body description.}
\column{0.47\linewidth}
\Card{5.6cm}{CardBg}{CardRule}{%
  \sffamily\footnotesize
  \textbf{What ALIGNN adds}\\[3pt]
  The \emph{angles} between bonds --- a three-body term. Resistance to shear
  depends on how bonds resist \emph{bending}, not only stretching.}
\end{columns}

\Takeaway{Shear modulus is a bond-bending property. A model blind to angles has
to infer it indirectly.}
\end{frame}

\begin{frame}
\SlideHeader{07 \textperiodcentered\ ALIGNN}{One crystal becomes \emph{two} graphs}

\begin{center}
\includegraphics[width=0.84\linewidth]{figures/alignn_line_graph.png}
\end{center}

\vspace{-4pt}
\begin{columns}[T]
\column{0.32\linewidth}
{\sffamily\scriptsize
\textbf{Neighbours.} A periodic search from the CIF: within \textbf{5.0 \AA},
at most \textbf{12} per atom (the trained model's config). Atoms start from
CGCNN's 92-number element table.\par}
\column{0.32\linewidth}
{\sffamily\scriptsize
\textbf{B: atom graph $\mathcal{G}$.} Atoms are nodes, bonds are edges carrying
the length $r_{ij}$. This is what CGCNN builds, and where it stops.\par}
\column{0.32\linewidth}
{\sffamily\scriptsize
\textbf{C: line graph $\mathcal{L}$.} Every \emph{bond} becomes a node; two bonds
sharing an atom are joined by an edge carrying their angle, fed as
$\cos\theta_{jik}$ on a radial basis.\par}
\end{columns}

\Takeaway{An angle belongs to a \emph{pair} of bonds, so it needs a graph whose
nodes are bonds. That graph is $\mathcal{L}$.}
\end{frame}

\begin{frame}
\SlideHeader{09 \textperiodcentered\ ALIGNN}{One ALIGNN layer, two graphs}

\begin{columns}[T]
\column{0.52\linewidth}
{\sffamily\footnotesize
One ALIGNN layer is two coupled updates, run in order:}

\vspace{5pt}
\begin{tikzpicture}[node distance=4.5mm,
  box/.style={rounded corners=4pt, fill=CardBg, draw=CardRule, line width=0.7pt,
              text width=5.6cm, align=left, inner sep=6pt,
              font=\sffamily\scriptsize\color{SlideInk}},
  arr/.style={font=\color{SlideBlue}\small}]
  \node[box] (b1) {\textbf{1. Update on the line graph $\mathcal{L}$}\\
    Each bond-node is updated from the bonds it shares an atom with, using the
    \emph{angle} on the connecting edge. Bond features now carry angular
    information.};
  \node[arr, below=of b1] (a1) {$\downarrow$};
  \node[box, below=of a1] (b2) {\textbf{2. Update on the atom graph $\mathcal{G}$}\\
    Each atom-node is updated from its neighbours using those
    \emph{angle-aware} bond features as edge inputs.};
\end{tikzpicture}

\column{0.44\linewidth}
\Card{5.4cm}{CardBg}{CardRule}{%
  \sffamily\footnotesize
  \textbf{Then, as in CGCNN}\\[3pt]
  Average the atom vectors over the cell, pass them through a small
  fully-connected head, and emit one number.}

\vspace{4pt}
\AccentCard{5.4cm}{SlideAmber}{%
  \sffamily\footnotesize
  \textbf{Trained on GPa, scored in $\log_{10}$}\\[3pt]
  ALIGNN is trained on the modulus itself, in GPa, with a mean-squared-error
  loss; its $\log_{10}$ errors are computed afterwards (predictions are floored
  at $10^{-3}$ GPa). CGCNN trains on $\log_{10}$ directly. Both train one
  network per modulus.\par}
\end{columns}
\end{frame}

\begin{frame}
\SlideHeader{15 \textperiodcentered\ PHYSICS}{From $K$, $G$ to sound speed, $\Theta_D$ and $\gamma$}

\begin{columns}[T]
\column{0.49\linewidth}
{\small
\begin{align}
v_l &= \sqrt{\frac{K + \tfrac{4}{3}G}{\rho}} \\[1pt]
v_t &= \sqrt{\frac{G}{\rho}} \\[1pt]
v_s &= \left[\frac{1}{3}\left(\frac{1}{v_l^{3}} + \frac{2}{v_t^{3}}\right)\right]^{-1/3}
\end{align}}
\column{0.49\linewidth}
{\small
\begin{align}
\Theta_D &= \frac{h}{k_B}\left(\frac{3}{4\pi V}\right)^{1/3} v_s \\[1pt]
\nu &= \frac{(v_l/v_t)^2 - 2}{2\,(v_l/v_t)^2 - 2} \\[1pt]
\gamma &= \frac{3\,(1 + \nu)}{2\,(2 - 3\nu)}
\end{align}}
\end{columns}

\vspace{2pt}
{\sffamily\scriptsize
$K$, $G$ bulk and shear modulus \emph{(predicted, GPa)} \textperiodcentered\
$\rho$ density \emph{(CIF)} \textperiodcentered\ $v_l$, $v_t$, $v_s$
longitudinal, transverse, average sound speed \textperiodcentered\ $\nu$
Poisson's ratio \textperiodcentered\ $\gamma$ Gr\"uneisen parameter
\textperiodcentered\ $V$ volume of the CIF's \emph{primitive} cell, as in the
paper's released code (the textbook Debye formula uses volume per atom).

\vspace{3pt}
\textbf{Eq.~6 matters most}: $\gamma$ is \emph{derived} from the same $K$, $G$
--- the ``Poisson $\gamma$''.\par}
\end{frame}

\begin{frame}
\SlideHeader{16 \textperiodcentered\ PHYSICS}{The $\kappa_L$ formula, and ``the Slack chain''}

\vspace{-4pt}
{\small
\begin{align}
\kappa_L^{\text{Slack}} &=
  \frac{A(\gamma)\,\bar{M}\,V^{1/3}\,\Theta_D^{3}}{\gamma^{2}\,T\,N}
  &&\text{Slack's model, } A(\gamma) = \frac{2.43\times10^{-8}}{1 - 0.514/\gamma + 0.228/\gamma^{2}} \\[2pt]
\kappa_L^{\text{cal}} &= \frac{G\,v_s\,V^{1/3}}{N\,T}\;e^{-\gamma}
  &&\text{the boxed equation of the paper's Figure 1}
\end{align}}

{\sffamily\scriptsize
$\bar{M}$ mean atomic mass (amu; the released code puts in the total cell
mass), $N$ atoms per primitive cell, $T = 300$ K. No number here uses Eq.~7. \textbf{This project reports $\kappa_L^{\text{cal}}$}: changing $\gamma$
rescales it by $e^{\gamma_{\text{old}} - \gamma_{\text{new}}}$, which is how the
MLIP $\gamma$ is swapped in later.\par}

\vspace{3pt}
\begin{center}
\scalebox{0.80}{%
\begin{tikzpicture}[node distance=3mm,
  box/.style={rounded corners=4pt, fill=CardBg, draw=CardRule, line width=0.7pt,
              text width=2.2cm, align=center, inner sep=4pt,
              font=\sffamily\scriptsize\color{SlideInk}},
  arr/.style={font=\color{SlideBlue}\small}]
  \node[box, fill=BlueBg] (b1) {$K$, $G$\\ {\color{Muted}predicted}};
  \node[arr, right=of b1] (a1) {$\rightarrow$};
  \node[box, right=of a1] (b2) {sound speeds\\ Eqs.~1--3};
  \node[arr, right=of b2] (a2) {$\rightarrow$};
